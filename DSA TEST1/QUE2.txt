package p1;

public class pattern {

    public static void main(String[] args) {

        for (char i = 'A'; i <= 'E'; i++) {
            for (char j = i; j <= 'E'; j++) {
                System.out.print(j + " ");
            }
            System.out.println();
        }

        for (char i = 'D'; i >= 'A'; i--) {
            for (char j = i; j <= 'E'; j++) {
                System.out.print(j + " ");
            }
            System.out.println();
        }
    }
}
